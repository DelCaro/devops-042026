nodes = [
	{:name => "APP1", :ip => "172.16.10.10"},
	{:name => "APP2", :ip => "172.16.10.11"},
	{:name => "APP3", :ip => "172.16.10.12"},
]

Vagrant.configure("2") do |config|


nodes.each do |app_config|


config.vm.define app_config[:name] do |vm|

	vm.vm.box = "shekeriev/debian-12"
	vm.vm.hostname = app_config[:name]
	vm.vm.network "private_network", ip: app_config[:ip]
	
vm.vm.provider("virtualbox") do |vb|
	vb.name = app_config[:name]
	vb.memory = 2048
	vb.cpus = 2
	

end
end
end
end



#nodes é um array de hashes com os dados de cada VM.
#each itera sobre cada elemento do array.
#app_config é a variável que recebe cada elemento (um hash com :name e :ip).
